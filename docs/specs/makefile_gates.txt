```makefile
	python3 test/gate.py $(STEP2) $(STEP3)
```

`test/gate.py` — the decision, and the only thing that authorizes proceeding.

```python
#!/usr/bin/env python3
# SPDX-License-Identifier: MIT
import json, sys
AREA_MAX, T1_MAX, T2_MAX = 3.60, 4.5, 4.5
a = json.load(open("build/area.json"))["ratio"]
t1 = json.load(open(sys.argv[1]))["peak"]
t2 = json.load(open(sys.argv[2]))["peak"]

print(f"area {a:.2f}x (max {AREA_MAX})")
print(f"|t|1st {t1:.2f} (max {T1_MAX})")
print(f"|t|2nd {t2:.2f} (max {T2_MAX})")

if t2 >= T2_MAX:
    print("\nVERDICT: 2 shares insufficient -> build 3-share variant")
    sys.exit(2)
if t1 >= T1_MAX:
    print("\nVERDICT: first-order leak -> masked CSA incomplete")
    sys.exit(3)
if a > AREA_MAX:
    print("\nVERDICT: area over budget -> shared multiplier")
    sys.exit(4)
print("\nVERDICT: 2-share accepted")
```

Exit codes matter — `2` triggers the 3-share branch, `4` triggers the shared-multiplier branch. The build does not proceed to tapeout on any other code.

`tools/fst_acc.py` — FST to weighted pJ trace, using the model.

```python
#!/usr/bin/env python3
# SPDX-License-Identifier: MIT
import json, sys
import numpy as np
from pyfst import FstReader          # thin ctypes binding, see tools/pyfst.py

def acc(fst_path, model_path, out, window, vdd=0.9):
    m = json.load(open(model_path))
    caps, slew = m["caps"], m["slew"]
    r = FstReader(fst_path)
    code_cap = np.zeros(r.var_count + 1)
    code_slew = np.full(r.var_count + 1, 60.0)
    for code, name in r.vars():
        c = caps.get(name)
        if c:
            code_cap[code] = c
            code_slew[code] = slew.get(name, 60.0)
    trace = np.zeros(window)
    for t, changes in r.iter_blocks():
        for code, _v in changes:
            if code_cap[code]:
                cyc = min(t // 2, window - 1)
                trace[cyc] += 0.5 * code_cap[code] * vdd * vdd * 1e12
                trace[cyc] += 0.014 * code_cap[code] * code_slew[code]
    np.save(out, trace)
    print(f"{window} samples, mean {trace.mean():.3f} pJ")

if __name__ == "__main__":
    acc(sys.argv[1], "build/model.json", sys.argv[3], 4000)
```

`test/tvla_gate.py` (weighted version)

```python
#!/usr/bin/env python3
# SPDX-License-Identifier: MIT
import json, sys
import numpy as np
GATE = 4.5
tr = np.load(sys.argv[1])
n = len(tr) // 2
```

`test/poi_select.py` — top-64 samples by |t|, one per 8-sample window to avoid clustering.

```python
#!/usr/bin/env python3
# SPDX-License-Identifier: MIT
import sys
import numpy as np
tr = np.load(sys.argv[1]); n = len(tr) // 2
g0, g1 = tr[:n][12:], tr[n:][12:]
t = np.abs(g0.mean(0) - g1.mean(0)) / np.sqrt(

order = np.argsort(-t)
pois, used = [], np.zeros(len(t), bool)

    pois.append(int(i)); used[max(0,i-8):i+9] = True
    if len(pois) == 64: break
np.save(sys.argv[2], np.array(pois))
print(f"{len(pois)} POIs, min |t| {t[pois[-1]]:.2f}")
```

# Step 3 — Second order at 10⁶, streaming

1e6 traces × 4 000 samples × 8 bytes = 32 GB. Not holdable. Two fixes: accumulate only POIs, and merge Welford state across workers.

`tools/fst_acc2.py` — POI-only, centered-square, Welford merge.

```python
#!/usr/bin/env python3
# SPDX-License-Identifier: MIT
import json, sys
import numpy as np
from pyfst import FstReader

class Welford:
    """Parallel-mergeable running mean/M2 for second-order moments."""
    def __init__(self, k):
        self.n = 0; self.mean = np.zeros(k); self.m2 = np.zeros(k)
    def add(self, x):
        self.n += 1
        d = x - self.mean
        self.mean += d / self.n
        self.m2 += d * (x - self.mean)
    def merge(self, o):
        if o.n == 0: return
        n = self.n + o.n
        d = o.mean - self.mean
        self.mean = (self.mean * self.n + o.mean * o.n) / n
        self.m2 += o.m2 + d * d * self.n * o.n / n
        self.n = n

def run(fst_path, model_path, poi, vdd=0.9):
    m = json.load(open(model_path))
    caps, slew = m["caps"], m["slew"]
    r = FstReader(fst_path)
    code_cap = np.zeros(r.var_count + 1)
    for code, name in r.vars():
        c = caps.get(name)
        if c: code_cap[code] = c
    full = np.zeros(4000)
    for t, changes in r.iter_blocks():
        for code, _v in changes:
            if code_cap[code]:
                full[min(t // 2, 3999)] += 0.5 * code_cap[code] * vdd * vdd * 1e12
    sel = full[12:][poi]
    return sel

if __name__ == "__main__":
    fst, out, group = sys.argv[1], sys.argv[2], int(sys.argv[3])
    poi = np.load("captures/poi.npy")
    x = run(fst, "build/model.json", poi)
    np.save(out, np.stack([x, x * x]))
```

`test/tvla2_run.py` — mean pass then variance pass, parallel.

```python
#!/usr/bin/env python3
# SPDX-License-Identifier: MIT
import argparse, json, subprocess, sys
from concurrent.futures import ThreadPoolExecutor
import numpy as np
from tools.fst_acc2 import Welford, run

def worker(args):
    lo, hi, group = args
    w = Welford(64)
    for i in range(lo, hi):
        subprocess.run(["./obj_dir/tvla_stream", "1", f"/tmp/f{group}_{i}"],
                       check=True, capture_output=True)
        w.add(run(f"/tmp/f{group}_{i}", "build/model.json",
                  np.load("captures/poi.npy")))
    return w

def main():
    p = argparse.ArgumentParser()
    p.add_argument("--traces", type=int, default=1_000_000)
    p.add_argument("--poi", required=True)
    p.add_argument("-j", type=int, default=8)
    a = p.parse_args()
    half = a.traces // 2

    res = {}
    for g in (0, 1):
        chunk = half // a.j
        jobs = [(g*half + k*chunk, g*half + (k+1)*chunk, g) for k in range(a.j)]
        with ThreadPoolExecutor(a.j) as ex:
            ws = list(ex.map(worker, jobs))
        tot = Welford(64)
        for w in ws: tot.merge(w)
        res[g] = tot

    w0, w1 = res[0], res[1]
    # second-order statistic: E[(x-mu)^2], variance from M2 of squares
    s0, s1 = w0.m2 / w0.n, w1.m2 / w1.n
    v0 = (w0.m2 / w0.n) / w0.n
    v1 = (w1.m2 / w1.n) / w1.n
    t = (s0 - s1) / np.sqrt(v0 + v1)
    peak, idx = float(np.nanmax(np.abs(t))), int(np.nanargmax(np.abs(t)))
    json.dump({"peak": peak, "idx": idx, "n": int(w0.n)}, sys.stdout)
    print(f"\n2nd-order |t|max {peak:.2f} @ {idx}  n={w0.n}", file=sys.stderr)

main()
```

Runtime: gate-level at ~40 k cycles/s, 400 cycles/trace, 8 workers ≈ 640 traces/s → 1e6 in ~26 min. Feasible.

Noise floor at 1e6 is ~1.8. **Anything above 4.5 is structural, not noise.**

# Step 4 — 3-share fallback

Correct the arithmetic first: 2-share needs 4 truncated multipliers, 3-share needs 9. So 3-share is ~2.25× the masked area, not 2×. With a shared multiplier it is 9 cycles/MAC.

`rtl/mont_mul_masked3.v` — shared multiplier, 3 shares, DOM-style AND with 3 fresh randoms.

```verilog
// SPDX-License-Identifier: MIT
`timescale 1ns/1ps

module mont_mul_masked3 (
    input  wire         clk, rst_n, start,
    input  wire [255:0] a0, a1, a2, b0, b1, b2, modulus,
    input  wire [31:0]  n0prime,
    input  wire [31:0]  rnd,
    output reg  [255:0] r0, r1, r2,
    output reg          done
);
    // 3-share ISW: 9 products, 3 randoms per MAC, shared multiplier
    reg [255:0] ta0, ta1, ta2, tb0, tb1, tb2, tm;
    reg [31:0]  tn0;
    always @(*) begin
        ta0 = a0; ta1 = a1; ta2 = a2;
        tb0 = b0; tb1 = b1; tb2 = b2; tm = modulus; tn0 = n0prime;
    end

    wire [31:0] ai = ta0[32*i +: 32] ^ ta1[32*i +: 32] ^ ta2[32*i +: 32];
    wire [31:0] bj = tb0[32*j +: 32] ^ tb1[32*j +: 32] ^ tb2[32*j +: 32];
    wire [31:0] mj = tm[32*j +: 32];

    reg [31:0] p [0:8];
    reg [31:0] r1_q, r2_q, r3_q;
    reg [3:0]  k;
    reg [2:0]  i, j;
    reg [1:0]  st;
    // 3 masked accumulator shares, carry-save
    reg [255:0] s0, s1, s2, c0, c1, c2;
    reg [31:0]  m0, m1, m2;

    wire [31:0] sv = s0[32*j+:32] ^ s1[32*j+:32] ^ s2[32*j+:32];
    wire [31:0] cv = c0[32*j+:32] ^ c1[32*j+:32] ^ c2[32*j+:32];
    wire [31:0] zv = m0 ^ m1 ^ m2;
    wire [31:0] sum_v = sv ^ cv ^ zv;
    wire [31:0] car_v = (sv & cv) | (sv & zv) | (cv & zv);

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin st <= 0; done <= 0; k <= 0; end
        else begin
            done <= 0;
            case (st)
                0: if (start) begin
                       s0 <= 0; s1 <= 0; s2 <= 0; c0 <= 0; c1 <= 0; c2 <= 0;
                       i <= 0; j <= 0; k <= 0; st <= 1;
                   end
                1: begin                        // one shared product per cycle
                       case (k)
                           0: p[0] <= ai * bj; 1: p[1] <= ai * bj;
                           2: p[2] <= ai * bj; 3: p[3] <= ai * bj;
                           4: p[4] <= ai * bj; 5: p[5] <= ai * bj;
                           6: p[6] <= ai * bj; 7: p[7] <= ai * bj;
                           8: p[8] <= ai * bj;
                       endcase
                       r1_q <= rnd; r2_q <= rnd; r3_q <= rnd;
                       if (k == 8) st <= 2; else k <= k + 1;
                   end
                2: begin                        // ISW recombine, 3 randoms
                       m0 <= p[0] ^ p[1] ^ r1_q;
                       m1 <= p[2] ^ p[3] ^ p[4] ^ r1_q ^ r2_q;
                       m2 <= p[5] ^ p[6] ^ p[7] ^ p[8] ^ r2_q ^ r3_q;
                       st <= 3;
                   end
                3: begin
                       s0[32*j+:32] <= sum_v ^ m0; s1[32*j+:32] <= m1;
                       s2[32*j+:32] <= m2;
                       c0[32*j+:32] <= car_v ^ r3_q; c1[32*j+:32] <= r1_q;
                       c2[32*j+:32] <= r2_q;
                       k <= 0;
                       if (j == 7) begin
                           i <= i + 1; j <= 0; st <= (i == 7) ? 4 : 1;
                       end else begin j <= j + 1; st <= 1; end
                   end
                4: begin r0 <= s0 ^ c0; r1 <= s1 ^ c1; r2 <= s2 ^ c2;
                          done <= 1; st <= 0; end
            endcase
        end
    end
endmodule
```

Latency 12 cycles/MAC → 1 538 per multiply, 6× the 2-share. Area ~2.6× plain. Randomness 3 words/MAC.

# Step 5 — Step 4 fixes (correctness, after the numbers land)

Only these, in this order:

1. `mm_dst` declaration + `step_mul` assignment.
2. Delete the dead `prod` line in `ecdsa_core` state 0; widen `kk` to 511 bits.
3. Doubling ×3: temp register, two `fp_add` on distinct operands.
4. `fp_reduce_n` Montgomery sequence; delete both `%`.
5. `ap_no_inf_midloop`, `ap_rs_nonzero`, `ap_fixed_iters` in the SVA bind.
6. `map_check.py` into the build.

None of these move |t|. That is exactly why they wait.

`test/policy_negative.py` — one negative case per predicate, each must deny.

```python
#!/usr/bin/env python3
# SPDX-License-Identifier: MIT
import json, sys
cases = json.load(open(sys.argv[1]))["cases"]
EXPECT = {"weight", "pcr", "deployment", "platform", "chain", "nonce", "counter"}
seen = set()
bad = []
for c in cases:
    if c["expect"] != "deny":
        bad.append(f"{c['name']}: expected deny, got {c['verdict']}")
    for p in c["broken"]:
        seen.add(p)
missing = EXPECT - seen
if missing: bad.append(f"predicates untested: {sorted(missing)}")
for b in bad: print(b)
print("policy suite", "PASS" if not bad else "FAIL")
sys.exit(0 if not bad else 1)
```

Seven predicates, seven negative cases. A predicate with no negative test is a predicate that has never been shown to fire.

## What the numbers mean at each gate

| Exit | Meaning | Action |
|---|---|---|
| 0 | 2-share holds, area in budget | proceed to correctness work |
| 2 | second-order leak | 3-share, re-run steps 2–3, re-budget |
| 3 | first-order leak | masked CSA is wrong, not the share count |
| 4 | area over budget | shared multiplier, accept latency, re-run step 3 |

The 3-share branch is not a failure of the design, it is a known cost, and the only reason to run step 3 before the floorplan freeze is that the 2.6× figure has to be reserved *now* rather than discovered after placement.
